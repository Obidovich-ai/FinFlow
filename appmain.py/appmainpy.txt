from fastapi import FastAPI, Depends
from .database import SessionLocal, User, Transaction
from .payme_service import PaymeService

app = FastAPI()
payme = PaymeService(merchant_id="YOUR_ID", secret_key="YOUR_KEY")

@app.post("/pay/create")
def make_payment(user_id: int, sum: float):
    db = SessionLocal()
    # 1. Bazada yangi tranzaksiya ochish
    new_tx = Transaction(user_id=user_id, amount=sum, status="pending", provider="payme")
    db.add(new_tx)
    db.commit()
    
    # 2. To'lov havolasini yaratish
    link = payme.create_pay_link(sum, new_tx.id)
    return {"pay_url": link}
from fastapi.middleware.cors import CORSMiddleware

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"], # Barcha qurilmalarga ruxsat berish
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)
